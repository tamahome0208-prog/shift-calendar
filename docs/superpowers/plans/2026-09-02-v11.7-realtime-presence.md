```javascript
// v11.7 presenceStore: 位置/セル/active を throttle して write
const presenceStore = {
  THROTTLE_MS: 2000,
  IDLE_MS: 5000,
  ENABLED_KEY: 'sc_partner_cursor_enabled',
  buffer: { x: 0, y: 0, cell: null, active: true },
  lastWriteTs: 0,
  idleTimer: null,
  pendingTimer: null,

  isEnabled() {
    const v = localStorage.getItem(this.ENABLED_KEY);
    return v === null ? true : v === 'true';
  },
  setEnabled(v) {
    localStorage.setItem(this.ENABLED_KEY, v ? 'true' : 'false');
    if (!v) this.setActive(false);
  },

  updatePosition(x, y, cellKey) {
    if (!this.isEnabled()) return;
    const me = identityStore.get();
    if (!me || !firestoreDb) return;
    this.buffer.x = x;
    this.buffer.y = y;
    this.buffer.cell = cellKey || null;
    this.buffer.active = true;
    const now = Date.now();
    // idle timer 更新
    if (this.idleTimer) clearTimeout(this.idleTimer);
    this.idleTimer = setTimeout(() => this.setActive(false), this.IDLE_MS);
    // throttle write
    if (now - this.lastWriteTs >= this.THROTTLE_MS) {
      this._flushNow();
    } else if (!this.pendingTimer) {
      const delay = this.THROTTLE_MS - (now - this.lastWriteTs);
      this.pendingTimer = setTimeout(() => { this.pendingTimer = null; this._flushNow(); }, delay);
    }
  },
  setActive(v) {
    if (!this.isEnabled() && v) return;
    if (this.buffer.active === v) return;
    this.buffer.active = v;
    if (!v && this.idleTimer) { clearTimeout(this.idleTimer); this.idleTimer = null; }
    this._flushNow();
  },
  _flushNow() {
    const me = identityStore.get();
    if (!me || !firestoreDb) return;
    this.lastWriteTs = Date.now();
    Storage.writePresence({ ...this.buffer }).catch(e => console.error('[presence] write failed:', e));
  },
};
```

```javascript
const PartnerCursor = {
  MAX_AGE_MS: 30000,

  update() {
    const me = identityStore.get();
    if (!me) return this._hide();
    if (!presenceStore.isEnabled()) return this._hide();
    // 相手の presence を探す(自分以外の最初のエントリ)
    const partnerKey = me === 'kouki' ? 'yui' : 'kouki';
    const partner = firestorePresence[partnerKey];
    if (!partner || !partner.active) return this._hide();
    // 古い state は無視
    const ts = partner.lastPing?.toMillis ? partner.lastPing.toMillis() : 0;
    if (Date.now() - ts > this.MAX_AGE_MS) return this._hide();
    this._show(partner, partnerKey);
  },
  _show(data, partnerKey) {
    const el = document.getElementById('partner-cursor');
    if (!el) return;
    const label = document.getElementById('partner-cursor-name');
    if (label) label.textContent = partnerKey === 'kouki' ? 'こうき' : 'ゆい';
    const color = partnerKey === 'kouki' ? 'var(--kouki-meeting)' : 'var(--yui-meeting)';
    el.style.setProperty('--partner-color', color);
    // 0-1 相対座標を絶対 px に変換
    const px = data.x * window.innerWidth;
    const py = data.y * window.innerHeight;
    el.style.transform = `translate(${px}px, ${py}px)`;
    el.style.display = 'block';
    // 相手が hover 中のセルにアウトライン
    document.querySelectorAll('.cell.partner-hover').forEach(c => c.classList.remove('partner-hover'));
    if (data.cell) {
      const cell = document.querySelector(`.cell[data-key="${data.cell}"]`);
      if (cell) cell.classList.add('partner-hover');
    }
  },
  _hide() {
    const el = document.getElementById('partner-cursor');
    if (el) el.style.display = 'none';
    document.querySelectorAll('.cell.partner-hover').forEach(c => c.classList.remove('partner-hover'));
  },
};
```

```javascript
const OverlapDetector = {
  HOLD_MS: 2000,
  COOLDOWN_MS: 10000,
  currentMyCell: null,
  matchStartTs: 0,
  lastFireTs: 0,

  onMyCellChange(cellKey) {
    if (this.currentMyCell === cellKey) return;
    this.currentMyCell = cellKey;
    this.matchStartTs = 0;  // リセット
  },

  check() {
    const me = identityStore.get();
    if (!me) return;
    const partnerKey = me === 'kouki' ? 'yui' : 'kouki';
    const partner = firestorePresence[partnerKey];
    if (!partner || !partner.active || !partner.cell) {
      this.matchStartTs = 0;
      return;
    }
    if (this.currentMyCell !== partner.cell) {
      this.matchStartTs = 0;
      return;
    }
    // cooldown 中はスキップ
    if (Date.now() - this.lastFireTs < this.COOLDOWN_MS) return;
    // 一致中、タイマー開始
    if (this.matchStartTs === 0) {
      this.matchStartTs = Date.now();
      return;
    }
    if (Date.now() - this.matchStartTs >= this.HOLD_MS) {
      this.lastFireTs = Date.now();
      this.matchStartTs = 0;
      spawnOverlapCelebration(this.currentMyCell);
    }
  },
};
```
